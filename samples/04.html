function setupOcclusionQuery() {
  // Initialize query object
  const query = gl.createQuery();
  
  // Begin query
  gl.beginQuery(gl.ANY_SAMPLES_PASSED, query);
  // Render occluder geometry
  renderOccluder();
  gl.endQuery(gl.ANY_SAMPLES_PASSED);
  
  // Check results later
  const result = gl.getQueryParameter(query, gl.QUERY_RESULT_AVAILABLE);
}
